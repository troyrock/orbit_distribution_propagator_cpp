project(distribution_propagator VERSION 0.1.0 LANGUAGES CXX)

  endif()
endif()

# Windows searches beside the executable before PATH. Deploy only the matching
# compiler's runtime DLLs; an older Meld/MSYS pthread DLL can lack nanosleep64.
if(WIN32 AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  get_filename_component(_compiler_bin "${CMAKE_CXX_COMPILER}" DIRECTORY)
  set(_gcc_runtime_files "")
  foreach(dll libstdc++-6.dll libgcc_s_seh-1.dll libwinpthread-1.dll)
    if(NOT EXISTS "${_compiler_bin}/${dll}")
      message(FATAL_ERROR "Matching compiler runtime missing: ${_compiler_bin}/${dll}")
    endif()
    list(APPEND _gcc_runtime_files "${_compiler_bin}/${dll}")
  endforeach()
  foreach(target distribution_propagator distribution_coverage distribution_coverage_tests distribution_phase_law_tests distribution_math_tests distribution_config_tests distribution_backend_tests distribution_backend_probe distribution_orekit_fixture)
    if(TARGET ${target})
      add_custom_command(TARGET ${target} POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different ${_gcc_runtime_files} "$<TARGET_FILE_DIR:${target}>"
        COMMAND_EXPAND_LISTS VERBATIM)
    endif()
  endforeach()
endif()
